<script>
    // Station information is supplied by Flask.
    const stations = {{ map_stations | tojson }};

    // Default map view if no station coordinates are available.
    const map = L.map("weatherMap").setView([20.5937, 78.9629], 5);

    L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
        maxZoom: 18,
        attribution: '&copy; OpenStreetMap contributors'
    }).addTo(map);

    const bounds = [];

    stations.forEach(function(station) {
        const latitude = Number(station.latitude);
        const longitude = Number(station.longitude);

        if (!Number.isFinite(latitude) ||
            !Number.isFinite(longitude)) {
            return;
        }

        const marker = L.circleMarker([latitude, longitude], {
            radius: 10,
            color: station.risk_color,
            fillColor: station.risk_color,
            fillOpacity: 0.85,
            weight: 2
        }).addTo(map);

        // Construct popup content safely using textContent.
        const popup = document.createElement("div");

        const heading = document.createElement("strong");
        heading.textContent = station.station_name || "Weather Station";
        popup.appendChild(heading);

        const region = document.createElement("p");
        region.textContent = "Region: " + (station.region_name || "Not specified");
        popup.appendChild(region);

        const temperature = document.createElement("p");
        temperature.textContent =
            "Temperature: " +
            (station.temperature !== null
                ? Number(station.temperature).toFixed(1) + " °C"
                : "N/A");
        popup.appendChild(temperature);

        const humidity = document.createElement("p");
        humidity.textContent =
            "Humidity: " +
            (station.humidity !== null
                ? Number(station.humidity).toFixed(1) + "%"
                : "N/A");
        popup.appendChild(humidity);

        const risk = document.createElement("p");
        risk.textContent = "Risk Level: " + station.risk;
        risk.style.fontWeight = "bold";
        risk.style.color = station.risk_color;
        popup.appendChild(risk);

        marker.bindPopup(popup);
        bounds.push([latitude, longitude]);
    });

    // Zoom to the available station locations.
    if (bounds.length > 0) {
        map.fitBounds(bounds, {
            padding: [35, 35],
            maxZoom: 12
        });
    }
</script>
